!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
!!!             WriteDigit                       !!!
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!   
    
    PROC WriteDigit(
        num board,
        num row,
        num col,
        num digit)

        VAR num cellSize;
        VAR num boardX;
        VAR num boardY;
        VAR num cx;
        VAR num cy;

        IF row < 1 OR row > gridSize OR
           col < 1 OR col > gridSize THEN
            TPWrite "Error: row or column out of range";
            RETURN;
        ENDIF

        IF digit < 1 OR digit > 4 THEN
            TPWrite "Error: digit must be 1-4";
            RETURN;
        ENDIF

        IF board = 1 THEN
            cellSize := largeCellSize;
            boardX := largeBoardX;
            boardY := largeBoardY;
        ELSEIF board = 2 THEN
            cellSize := smallCellSize;
            boardX := smallBoardX;
            boardY := smallBoardY;
        ELSE
            TPWrite "Error: board must be 1 or 2";
            RETURN;
        ENDIF

        cx := boardX + (col - 0.5) * cellSize;
        cy := boardY - (row - 0.5) * cellSize;

        TPWrite "Board: "\Num:=board;
        TPWrite "Row: "\Num:=row;
        TPWrite "Column: "\Num:=col;
        TPWrite "Digit: "\Num:=digit;
        TPWrite "Cell X: "\Num:=cx;
        TPWrite "Cell Y: "\Num:=cy;

        ! Select the drawing procedure for the requested digit
        IF digit = 1 THEN
            DrawDigit1 cx, cy, cellSize;
        ELSEIF digit = 2 THEN
            DrawDigit2 cx, cy, cellSize;
        ELSEIF digit = 3 THEN
            DrawDigit3 cx, cy, cellSize;
        ELSEIF digit = 4 THEN
            DrawDigit4 cx, cy, cellSize;
        ELSE
            TPWrite "Digit must be between 1 and 4";
            RETURN;
        ENDIF
    ENDPROC

 PROC WriteDigitAt(
    num cx,
    num cy,
    num cellSize,
    num digit)

    IF digit < 1 OR digit > 4 THEN
        TPWrite "Error: invalid digit";
        RETURN;
    ENDIF

    IF digit = 1 THEN
        DrawDigit1 cx, cy, cellSize;
    ELSEIF digit = 2 THEN
        DrawDigit2 cx, cy, cellSize;
    ELSEIF digit = 3 THEN
        DrawDigit3 cx, cy, cellSize;
    ELSEIF digit = 4 THEN
        DrawDigit4 cx, cy, cellSize;
    ENDIF

ENDPROC

!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
!!!           ProcessCommand                     !!!
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!

 
    PROC ProcessCommand(string command)
    
        VAR num board;
        VAR num row;
        VAR num col;
        VAR num digit;
        VAR bool conversionOK;
    
        TPWrite "1: Command received";
        TPWrite "Received: " + command;
    
        ! Handle PING first
        IF command = "PING" THEN
    
            TPWrite "PING received from Python";
    
            SocketSend pcSocket \Str:="PONG\0A";
    
            TPWrite "PONG sent to Python";
    
            RETURN;
    
        ENDIF
        
        
        ! Handle PARK
        IF command = "PARK" THEN
        
            TPWrite "PARK command received";
        
            GoToPark;
        
            SocketSend pcSocket \Str:="OK\0A";
        
            TPWrite "Robot parked";
        
            RETURN;
        
        ENDIF
        
        
        ! Handle DONE
        IF command = "DONE" THEN
        
            TPWrite "Sudoku board completed";
        
            GoToPark;
        
            SocketSend pcSocket \Str:="OK\0A";
        
            TPWrite "Robot returned to parking position";
        
            RETURN;
        
        ENDIF
    
        ! Check command type
        IF StrPart(command,1,5) <> "WRITE" THEN
            TPWrite "ERROR: Invalid command";
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Check command length
        IF StrLen(command) <> 13 THEN
            TPWrite "ERROR: Invalid command length";
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Check spaces
        IF StrPart(command,6,1) <> " " OR
           StrPart(command,8,1) <> " " OR
           StrPart(command,10,1) <> " " OR
           StrPart(command,12,1) <> " " THEN
    
            TPWrite "ERROR: Invalid spaces";
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Convert board ID
        conversionOK := StrToVal(StrPart(command,7,1), board);
        IF NOT conversionOK THEN
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Convert row
        conversionOK := StrToVal(StrPart(command,9,1), row);
        IF NOT conversionOK THEN
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Convert column
        conversionOK := StrToVal(StrPart(command,11,1), col);
        IF NOT conversionOK THEN
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        ! Convert digit
        conversionOK := StrToVal(StrPart(command,13,1), digit);
        IF NOT conversionOK THEN
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        TPWrite "2: Values converted";
    
        ! Validate values
        IF board < 1 OR board > 2 OR
           row < 1 OR row > 4 OR
           col < 1 OR col > 4 OR
           digit < 1 OR digit > 4 THEN
    
            TPWrite "ERROR: Values out of range";
            SocketSend pcSocket \Str:="ERROR\0A";
            RETURN;
        ENDIF
    
        TPWrite "Board: " \Num:=board;
        TPWrite "Row: " \Num:=row;
        TPWrite "Column: " \Num:=col;
        TPWrite "Digit: " \Num:=digit;
    
        TPWrite "3: Starting WriteDigit";
    
        ! Execute robot movement
        WriteDigit board, row, col, digit;
    
        TPWrite "4: WriteDigit completed";
    
        ! Send acknowledgement
        SocketSend pcSocket \Str:="OK\0A";
    
        TPWrite "5: OK sent";
    
    ENDPROC

    PROC DrawDigit1(num cx, num cy, num cellSize)
        VAR num s;
        VAR robtarget pStart;
        VAR robtarget pJoin;
        VAR robtarget pEnd;
    
        s := cellSize * 0.20;
    
        ! Start at the upper-left end of the diagonal

                   [0.029043357,0.000895114,0.999577752,-0.000026131],
                   [0,1,-2,1],
                   [9E9,9E9,9E9,9E9,9E9,9E9]];
    
        ! Join point at the top of the vertical stroke
        pJoin := [[cx, cy+s*1.35, penLift],
                  [0.029043357,0.000895114,0.999577752,-0.000026131],
                  [0,1,-2,1],
                  [9E9,9E9,9E9,9E9,9E9,9E9]];
    
        ! Bottom of the vertical stroke
        pEnd := [[cx, cy-s*1.35, penLift],
                 [0.029043357,0.000895114,0.999577752,-0.000026131],
                 [0,1,-2,1],
                 [9E9,9E9,9E9,9E9,9E9,9E9]];

        MoveJ pStart, V40, fine, toolPen\WObj:=wobjA3;
    
        ! Lower pen
        MoveL Offs(pStart,0,0,-penLift),
              V40, fine, toolPen\WObj:=wobjA3;
    
        ! Draw diagonal towards the vertical stroke
        MoveL Offs(pJoin,0,0,-penLift),
              V40, fine, toolPen\WObj:=wobjA3;

        MoveL Offs(pEnd,0,0,-penLift),
              V40, fine, toolPen\WObj:=wobjA3;

        MoveL pEnd, V40, fine, toolPen\WObj:=wobjA3;
    ENDPROC

!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!

    PROC WriteBoard(num board)
        VAR num row;
        VAR num col;
        VAR num digit;

        FOR row FROM 1 TO gridSize DO
            FOR col FROM 1 TO gridSize DO

                digit := sudokuBoard{row,col};

                IF digit <> 0 THEN
                    WriteDigit board,row,col,digit;
                ENDIF

            ENDFOR
        ENDFOR
    ENDPROC

    PROC SetCell(num board, num row, num col, num digit)
        IF board < 1 OR board > 2 THEN
            TPWrite "Error: board must be 1 or 2";
            RETURN;
        ENDIF

        IF row < 1 OR row > gridSize OR
           col < 1 OR col > gridSize THEN
            TPWrite "Error: row or column out of range";
            RETURN;
        ENDIF

        IF digit < 0 OR digit > 4 THEN
            TPWrite "Error: digit must be 0-4";
            RETURN;
        ENDIF

        sudokuBoard{row,col} := digit;

        TPWrite "Sudoku cell updated";
    ENDPROC


!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
!!!                RecieveCommand                !!!
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
    
    
    PROC ReceiveCommand()
    
        VAR bool messageComplete;
    
        receivedCommand := "";
        receivedChar := "";
        messageComplete := FALSE;
    
        WHILE NOT messageComplete DO
    
            SocketReceive pcSocket
                \Str:=receivedChar
                \ReadNoOfBytes:=1;
    
            ! Check for newline
            IF receivedChar = "\0A" THEN
    
                messageComplete := TRUE;
    
            ELSEIF receivedChar <> "\0D" THEN
    
                receivedCommand :=
                    receivedCommand + receivedChar;
    
            ENDIF
    
            ! Prevent excessively long messages
            IF StrLen(receivedCommand) > 30 THEN
                TPWrite "Command too long";
                messageComplete := TRUE;
                receivedCommand := "";
            ENDIF
    
        ENDWHILE
    
    ENDPROC

!!!                MAIN                          !!!
!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!! 
    
    
    
    PROC main()
    
        TPWrite "Starting TCP connection";
    
        SocketCreate pcSocket;
    
        SocketConnect pcSocket, PC_IP, PC_PORT;
    
        TPWrite "Connected to Python";
    
        ! Move robot outside camera view
        GoToPark;
    
        WHILE TRUE DO
    
            ReceiveCommand;
    
            IF StrLen(receivedCommand) > 0 THEN
                ProcessCommand receivedCommand;
            ENDIF
    
        ENDWHILE
    
    ENDPROC
